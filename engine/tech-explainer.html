class CompoundingScorer:
    """Station ④: score → verify → reshape centroids.
    ~844 parameters. No LLM. No GPU. O(1) per decision."""

    def score(self, factors: List[float], category: str) -> ScoreResult:
        """Softmax over L2 distances to category centroids."""
        v = np.array(factors)
        μ = self._graph_store.get_centroids(category)
        d_sq = {a: np.sum((v - μ[category][a]) ** 2) for a in self._actions}
        logits = {a: -d_sq[a] / self._tau for a in self._actions}  # τ = 0.1
        max_l = max(logits.values())
        exp_l = {a: np.exp(logits[a] - max_l) for a in self._actions}
        Z = sum(exp_l.values())
        probs = {a: exp_l[a] / Z for a in self._actions}
        best = max(probs, key=probs.get)
        return ScoreResult(action=best, confidence=probs[best],
                           basis=self._graph_store.get_similar(v, category, limit=3))

    def learn(self, factors, action_index, category, confirmed):
        """LVQ centroid update — conservation-gated."""
        η = 0.05 if confirmed else 0.01
        v = np.array(factors)
        centroids = self._graph_store.get_centroids()  # all categories
        μ_old = centroids[category][action_index]
        μ_new = μ_old + η * (v - μ_old)
        # μ₀ = μ_canonical + ε_firm; update is LVQ: μ ← μ + η(v - μ)
        new_centroids = centroids.copy()
        new_centroids[category][action_index] = μ_new
        gate = self._conservation_gate.check(centroids, new_centroids, category)
        if not gate.passed:
            return LearnResult(accepted=False, reason=gate.reason)
        centroids[category][action_index] = μ_new
        self._graph_store.save_centroids(centroids)
        return LearnResult(accepted=True, displacement=gate.displacement)
class SituationAnalyzer:
    """Station ②: assemble cross-system context → factor vector.
    Q-routed investigation — decides WHICH evidence and WHEN to stop."""

    async def assemble(self, entity_id: str) -> SituationContext:
        graph_ctx = await self._graph_store.get_context(entity_id)
        erp_ctx = await self._sap_connector.get_entity(entity_id)
        process_ctx = await self._celonis_connector.get_activities(entity_id)
        ctx = SituationContext(graph=graph_ctx, erp=erp_ctx, process=process_ctx)
        ctx.factors = self._extract_factors(ctx)
        ctx.category = self._classify(ctx.factors)
        return ctx

    def select_investigation(self, v, category) -> List[str]:
        """Q-routed: precision × leverage × discriminative.
        System reads the most informative evidence first."""
        q_values = {
            j: self._precision(j,v,category) * self._leverage(j,v,category)
               * self._discriminative(j,v,category)
            for j in self._available_factors
        }
        # Sorted by Q — top factors read first, stop when marginal Q drops
        return sorted(q_values, key=q_values.get, reverse=True)
class ConservationGate:
    """Station ⑧: three layers, all fixed at deployment approval.
    Learning cannot alter them. The gate protects the geometry."""

    def check(self, old_centroids, new_centroids, category) -> GateResult:
        α = self._current_phase()
        V = self.tensor_volume  # d × c × a

        # Layer 1: sufficient verified volume (α·q·V must exceed θ_min)
        if α * self._q * V < self.θ_min:
            return GateResult(passed=False, reason="insufficient_volume")

        # Displacement ceiling: centroid shift must stay within bounds
        disp = np.linalg.norm(new_centroids[category] - old_centroids[category])
        max_disp = 23.53 / (α * V)
        if disp > max_disp:
            return GateResult(passed=False, reason="displacement_exceeded")

        # Layer 2: per-factor displacement (no single factor drifts too far)
        for j in range(len(new_centroids[category])):
            δ_j = abs(new_centroids[category][j] - old_centroids[category][j])
            if δ_j > 0.7 * disp:  # m_rel = 0.7×
                return GateResult(passed=False, reason=f"per_factor_{j}")

        # Layer 3: rate of change (sudden shifts trigger pause)
        rate_short = self._rate(window=20)
        rate_long = self._rate(window=self._q)
        if rate_short > 1.15 * rate_long:  # short-window 15% above long = regime shift
            return GateResult(passed=False, reason="rate_of_change")

        return GateResult(passed=True, displacement=disp)
